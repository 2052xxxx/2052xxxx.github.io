JavaScript Solution Code

// Add click event listeners to all title bars
document.querySelectorAll('.title-bar').forEach(titleBar => {
  titleBar.addEventListener('click', function() {
    // Find the closest parent with a specific ID
    const parentWindow = this.closest('div[id^="window"]');
    
    // Remove highlight from all windows
    document.querySelectorAll('div[id^="window"]').forEach(window => {
      window.classList.remove('highlighted');
    });
    
    // If a parent window is found, highlight it and show feedback
    if (parentWindow) {
      parentWindow.classList.add('highlighted');
      document.getElementById('feedback').innerHTML = 
        `You clicked on the title bar of: ${parentWindow.id}`;
    }
  });
});
